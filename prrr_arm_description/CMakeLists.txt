cmake_minimum_required(VERSION 3.8)
project(prrr_arm_description)
if("$ENV{ROS_VERSION}" STREQUAL "1")
  find_package(catkin REQUIRED)
  catkin_package()
  install(DIRECTORY urdf meshes launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
  install(FILES README.zh-CN.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
elseif("$ENV{ROS_VERSION}" STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  install(DIRECTORY urdf meshes launch config DESTINATION share/${PROJECT_NAME})
  install(FILES README.zh-CN.md DESTINATION share/${PROJECT_NAME})
  ament_package()
else()
  message(FATAL_ERROR "Source your ROS environment first (ROS_VERSION must be 1 or 2).")
endif()
